def sync_core_rag_state():

    global faiss_index

    embedder = getattr(

    CORE_STATE["chunks"] = chunks

    CORE_STATE["embeddings"] = embeddings

# ============================================================
# EMBEDDER
# ============================================================

def load_embedder(
    status_callback=None
):

    global embedder

    try:

        if (
            embedder is not None
            and callable(
                getattr(
                    embedder,
                    "encode",
                    None
                )
            )
        ):

            return embedder

        if status_callback:

            status_callback(
                "Loading local embedding model..."
            )

        loaded = ingestion.load_embedder(
            status_callback=status_callback
        )

        # ----------------------------------------------------
        # load_embedder normally returns SentenceTransformer
        # ----------------------------------------------------

        if loaded is not None:

            if callable(
                getattr(
                    loaded,
                    "encode",
                    None
                )
            ):

                embedder = loaded

        # ----------------------------------------------------
        # Also check module-level state
        # ----------------------------------------------------

        if embedder is None:

            embedder = getattr(
                ingestion,
                "embedder",
                None
            )

        if embedder is None:

            raise RuntimeError(
                "Embedding model could not be loaded."
            )

        ingestion.embedder = embedder

        sync_core_rag_state()

        if status_callback:

            status_callback(
                "Embedding model ready."
            )

        return embedder

    except Exception as e:

        if status_callback:

            status_callback(
                f"Embedding model error: {e}"
            )

        print(
            "Embedding model error:",
            e
        )

        raise

def create_chunks(
    text,
    source,
    chunk_size=None,
    overlap=None
):

    function = getattr(
        document_processor,
        "create_chunks",
        None
    )

    # --------------------------------------------------------
    # Preferred modular implementation
    # --------------------------------------------------------

    if function is not None:

        try:

            if (
                chunk_size is not None
                and overlap is not None
            ):

                return function(
                    text,
                    source,
                    chunk_size=chunk_size,
                    overlap=overlap
                )

            return function(
                text,
                source
            )

        except TypeError:

            try:

                return function(
                    text,
                    source
                )

            except TypeError:

                return function(
                    text
                )

    # --------------------------------------------------------
    # Safe fallback chunker
    # --------------------------------------------------------

    if not text:

        return []

    text = str(text).strip()

    if not text:

        return []

    if chunk_size is None:

        chunk_size = 500

    if overlap is None:

        overlap = 80

    words = text.split()

    if not words:

        return []

    result = []

    start = 0

    while start < len(words):

        end = min(
            start + chunk_size,
            len(words)
        )

        chunk_text = " ".join(
            words[start:end]
        )

        if chunk_text:

            result.append(
                {
                    "source": source,
                    "text": chunk_text
                }
            )

        if end >= len(words):

            break

        start = max(
            end - overlap,
            start + 1
        )

    return result

# ============================================================
# CACHE
# ============================================================

def load_cached_memory(
    target_folder,
    status_callback=None
):

    global chunks
    global embeddings
    global faiss_index
    global embedder

    # --------------------------------------------------------
    # IMPORTANT:
    # core.ingestion requires a real state dictionary.
    # --------------------------------------------------------

    CORE_STATE["chunks"] = chunks

    CORE_STATE["embeddings"] = embeddings

    CORE_STATE["faiss_index"] = faiss_index

    CORE_STATE["embedder"] = embedder

    try:

        success = ingestion.load_cached_memory(
            target_folder=target_folder,
            status_callback=status_callback,
            state=CORE_STATE
        )

        if success:

            chunks = CORE_STATE.get(
                "chunks",
                []
            )

            embeddings = CORE_STATE.get(
                "embeddings"
            )

            faiss_index = CORE_STATE.get(
                "faiss_index"
            )

            embedder = CORE_STATE.get(
                "embedder",
                embedder
            )

            # ------------------------------------------------
            # Critical retrieval synchronization
            # ------------------------------------------------

            ingestion.chunks = chunks

            ingestion.embeddings = embeddings

            ingestion.faiss_index = faiss_index

            ingestion.embedder = embedder

            sync_core_rag_state()

        return bool(success)

    except Exception as e:

        print(
            "Cache load error:",
            e
        )

        if status_callback:

            status_callback(
                f"Cache load error: {e}"
            )

        return False


# ============================================================
# INGESTION
# ============================================================

def run_ingestion(
    target_folder,
    status_callback=None,
    stop_event=None
):

    global is_processing
    global chunks
    global embeddings
    global faiss_index
    global embedder

    if stop_event is None:

        stop_event = processing_stop_event

    if status_callback is None:

        status_callback = (
            lambda message: None
        )

    processing_stop_event.clear()

    is_processing = True

    try:

        # ----------------------------------------------------
        # Validate folder
        # ----------------------------------------------------

        if not target_folder:

            raise ValueError(
                "No document folder was selected."
            )

        if not os.path.isdir(
            target_folder
        ):

            raise FileNotFoundError(
                f"Document folder does not exist:\n"
                f"{target_folder}"
            )

        # ----------------------------------------------------
        # Load embedder
        # ----------------------------------------------------

        if (
            embedder is None
            or not callable(
                getattr(
                    embedder,
                    "encode",
                    None
                )
            )
        ):

            embedder = load_embedder(
                status_callback
            )

        # ----------------------------------------------------
        # Build complete state dictionary
        # ----------------------------------------------------

        CORE_STATE["chunks"] = chunks

        CORE_STATE["embeddings"] = embeddings

        CORE_STATE["faiss_index"] = faiss_index

        CORE_STATE["embedder"] = embedder

        # ----------------------------------------------------
        # Synchronize core before processing
        # ----------------------------------------------------

        push_core_rag_state()

        # ----------------------------------------------------
        # IMPORTANT:
        #
        # core.ingestion.run_ingestion() requires:
        #
        # state
        # load_embedder
        # extract_pdf
        # extract_docx
        # extract_txt
        # process_audio
        # create_chunks
        # embedder
        # ----------------------------------------------------

        success = ingestion.run_ingestion(

            target_folder=target_folder,

            status_callback=status_callback,

            stop_event=stop_event,

            state=CORE_STATE,

            load_embedder=load_embedder,

            extract_pdf=extract_pdf,

            extract_docx=extract_docx,

            extract_txt=extract_txt,

            process_audio=process_audio,

            create_chunks=create_chunks,

            embedder=embedder
        )

        # ----------------------------------------------------
        # Copy state back from modular core
        # ----------------------------------------------------

        chunks = CORE_STATE.get(
            "chunks",
            getattr(
                ingestion,
                "chunks",
                []
            )
        )

        embeddings = CORE_STATE.get(
            "embeddings",
            getattr(
                ingestion,
                "embeddings",
                None
            )
        )

        faiss_index = CORE_STATE.get(
            "faiss_index",
            getattr(
                ingestion,
                "faiss_index",
                None
            )
        )

        embedder = CORE_STATE.get(
            "embedder",
            getattr(
                ingestion,
                "embedder",
                embedder
            )
        )

        # ----------------------------------------------------
        # CRITICAL:
        # Retrieval.py reads ingestion.chunks directly.
        # ----------------------------------------------------

        ingestion.chunks = chunks

        ingestion.embeddings = embeddings

        ingestion.faiss_index = faiss_index

        ingestion.embedder = embedder

        sync_core_rag_state()

        if success:

            status_callback(
                "Ingestion complete: "
                f"{len(chunks)} chunks indexed."
            )

        return bool(success)

    except Exception as e:

        error_message = (
            f"Processing error: {e}"
        )

        print(
            error_message
        )

        try:

            status_callback(
                error_message
            )

        except Exception:
            pass

        return False

    finally:

        is_processing = False

# ============================================================
# STREAMING COMPATIBILITY
# ============================================================

def generate_stream(
    prompt,
    context="",
    **kwargs
):

    function = getattr(
        answer_engine,
        "generate_stream",
        None
    )

    if function is not None:

        return function(
            prompt,
            context,
            **kwargs
        )

    # --------------------------------------------------------
    # Fallback
    # --------------------------------------------------------

    answer = generate_answer(
        prompt,
        context,
        **kwargs
    )

    def fallback_generator():

        yield answer

    return fallback_generator()

                    "Folder selected. Ready to process."
                )

        threading.Thread(
            target=cache_worker,
            daemon=True
        ).start()


    # ========================================================
    # PROCESS DOCUMENTS
    # ========================================================

    def start_processing(
        self
    ):

        global is_processing

        if is_processing:

            return

        if not self.selected_folder:

            messagebox.showwarning(
                "No Folder",
                "Please select a document folder first."
            )

            return

        processing_stop_event.clear()

        is_processing = True

        self.process_button.configure(
            state="disabled",
            text="Processing..."
        )

        self.select_button.configure(
            state="disabled"
        )

        self.set_status(
            "Starting document ingestion..."
        )

        threading.Thread(
            target=self.processing_worker,
            daemon=True
        ).start()


    def processing_worker(
        self
    ):

        global is_processing

        try:

            def callback(
                message
            ):

                self.set_status(
                    message
                )

            success = run_ingestion(
                self.selected_folder,
                callback,
                processing_stop_event
            )

            sync_core_rag_state()

            if success:

                count = len(
                    chunks
                )

                vectors = (
                    faiss_index.ntotal
                    if faiss_index is not None
                    else 0
                )

                self.after(
                    0,
                    lambda: self.append_chat(
                        "SYSTEM",
                        (
                            "Document processing completed successfully.\n"
                            f"Chunks: {count}\n"
                            f"FAISS vectors: {vectors}"
                        )
                    )
                )

                self.set_status(
                    f"Ready â€¢ {count} chunks indexed"
                )

            else:

                self.set_status(
                    "Document processing failed."
                )

                self.after(
                    0,
                    lambda: messagebox.showerror(
                        "Processing Error",
                        (
                            "Document ingestion failed.\n\n"
                            "Check the terminal for details."
                        )
                    )
                )

        except Exception as e:

            print(
                "Processing worker error:",
                e
            )

            self.set_status(
                f"Processing error: {str(e)[:80]}"
            )

            self.after(
                0,
                lambda: messagebox.showerror(
                    "Processing Error",
                    str(e)
                )
            )

        finally:

            is_processing = False

            self.after(
                0,
                lambda: self.process_button.configure(
                    state="normal",
                    text="Process Documents"
                )
            )

            self.after(
                0,
                lambda: self.select_button.configure(
                    state="normal"
                )
            )


    # ========================================================
    # ENTER / SEND
    # ========================================================

    def handle_enter(
        self,
        event=None
    ):

        self.handle_action()

        return "break"


    def handle_action(
        self
    ):

        if is_generating:

            self.stop_generation()

            return

        self.start_generation()


    # ========================================================
    # STOP GENERATION
    # ========================================================

    def stop_generation(
        self
    ):

        global is_generating

        generation_stop_event.set()

        self.set_status(
            "Stopping generation..."
        )

        self.action_button.configure(
            state="disabled",
            text="Stopping..."
        )


    # ========================================================
    # START GENERATION
    # ========================================================

    def start_generation(
        self
    ):

        global is_generating
        global is_busy
        global last_response_text

        if is_generating:

            return

        query = (
            self.query_entry
            .get()
            .strip()
        )

        if not query:

            return

        if is_processing:

            messagebox.showwarning(
                "Processing",
                (
                    "Please wait until document processing "
                    "finishes."
                )
            )

            return

        # ----------------------------------------------------
        # Greetings do not require documents
        # ----------------------------------------------------

        greetings = {
            "hi",
            "hello",
            "hey",
            "hii",
            "hlo",
            "hello ai"
        }

        if query.lower() in greetings:

            self.query_entry.delete(
                0,
                "end"
            )

            self.append_chat(
                "USER",
                query
            )

            answer = (
                "Hello! I am your offline local AI assistant. "
                "Select and process documents to ask questions "
                "about your files."
            )

            self.append_chat(
                "ASSISTANT",
                answer
            )

            last_response_text = answer

            return

        # ----------------------------------------------------
        # Require indexed documents
        # ----------------------------------------------------

        sync_core_rag_state()

        if not chunks:

            messagebox.showwarning(
                "No Documents",
                (
                    "No documents are currently indexed.\n\n"
                    "Select a document folder and click "
                    "'Process Documents' first."
                )
            )

            return

        # ----------------------------------------------------
        # Start request
        # ----------------------------------------------------

        is_generating = True

        is_busy = True

        generation_stop_event.clear()

        self.query_entry.delete(
            0,
            "end"
        )

        self.query_entry.configure(
            state="disabled"
        )

        self.action_button.configure(
            text="Stop",
            state="normal"
        )

        self.append_chat(
            "USER",
            query
        )

        self.set_status(
            "Retrieving relevant documents..."
        )

        threading.Thread(
            target=self.generation_worker,
            args=(query,),
            daemon=True
        ).start()


    # ========================================================
    # GENERATION WORKER
    # ========================================================

    def generation_worker(
        self,
        query
    ):

        global is_generating
        global is_busy
        global last_response_text

        try:

            # ------------------------------------------------
            # Make sure retrieval sees current core state
            # ------------------------------------------------

            sync_core_rag_state()

            # ------------------------------------------------
            # RETRIEVAL
            # ------------------------------------------------

            selected, context = retrieve_context(
                query
            )

            if generation_stop_event.is_set():

                return

            if not selected or not context:

                answer = (
                    "I could not find this information "
                    "in the provided documents."
                )

                last_response_text = answer

                self.after(
                    0,
                    lambda: self.append_chat(
                        "ASSISTANT",
                        answer
                    )
                )

                self.set_status(
                    "No sufficiently relevant document found."
                )

                return

            sources = get_retrieved_sources(
                selected
            )

            self.set_status(
                f"Retrieved {len(selected)} relevant chunk(s)"
            )

            # ------------------------------------------------
            # GROUNDED PROMPT
            # ------------------------------------------------

            prompt = f"""
[INST]
You are an offline document question-answering assistant.

Answer the user's question using ONLY the provided local
document context.

STRICT RULES:

1. Do not use outside or pretrained knowledge.
2. Do not invent facts.
3. Do not guess.
4. If the answer is not supported by the context, say exactly:
   I could not find this information in the provided documents.
5. Give a direct and concise answer.
6. When useful, mention the source filename.
7. Do not create a different question.
8. Do not continue with unrelated information.

LOCAL DOCUMENT CONTEXT:
{context}

USER QUESTION:
{query}

ANSWER:
[/INST]
""".strip()

            # ------------------------------------------------
            # GENERATION
            # ------------------------------------------------

            self.set_status(
                "Generating answer with local Mistral..."
            )

            answer = generate_answer(
                prompt,
                "",
                max_tokens=MAX_OUTPUT_TOKENS,
                temperature=0.15,
                top_p=0.9,
                repeat_penalty=1.1
            )

            if generation_stop_event.is_set():

                return

            if not answer:

                answer = (
                    "I could not generate an answer from "
                    "the provided documents."
                )

            last_response_text = answer

            # ------------------------------------------------
            # Add source information only when appropriate.
            # ------------------------------------------------

            if sources:

                source_text = ", ".join(
                    sources
                )

                # Avoid duplicating source if model already
                # included it.

                if (
                    source_text.lower()
                    not in answer.lower()
                ):

                    answer = (
                        answer.rstrip()
                        + f"\n\nSource: {source_text}"
                    )

            self.after(
                0,
                lambda: self.append_chat(
                    "ASSISTANT",
                    answer
                )
            )

            self.set_status(
                "Ready"
            )

        except Exception as e:

            print(
                "Generation error:",
                e
            )

            error_text = (
                f"Generation error: {e}"
            )

            self.after(
                0,
                lambda: self.append_chat(
                    "SYSTEM",
                    error_text
                )
            )

            self.set_status(
                "Generation failed."

            )

        finally:

            is_generating = False

            is_busy = False

            generation_stop_event.clear()

            self.after(
                0,
                self.finish_generation
            )


    # ========================================================
    # FINISH GENERATION
    # ========================================================

    def finish_generation(
        self
    ):

        self.query_entry.configure(
            state="normal"
        )

        self.action_button.configure(
            state="normal",
            text="Send"
        )

        self.query_entry.focus_set()

        if not is_processing:

            self.set_status(
                "Ready"
            )
